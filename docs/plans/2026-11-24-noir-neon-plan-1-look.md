# Noir Neon — Plan 1: The Look — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give every screen of Coup the Noir Neon look (black, cyan `#00E5FF` and electric blue `#1E6BFF` glow, cyan haze, solid-colour character cards in a glowing cyan frame) without changing any layout, game logic, Firebase or multiplayer code.

**Architecture:** Nearly every colour in the app already comes from `src/theme.ts`, so most of the restyle is new theme tokens plus a handful of shared building blocks (`Screen` + new `Backdrop`, `Panel`, `Button`, `CharacterCard`, `CardBack`) and glow tweaks to the table parts (`Seat`, `ActionTile`, `EventBanner`, `TopBar`, `Sheet`). Two small decorative PNGs (background haze, card shading) are generated by a script. The Android launch window turns black.

**Tech Stack:** React Native 0.87 (new architecture), TypeScript, React's built-in `boxShadow` / `textShadow` styles, Jest + react-test-renderer, Python 3 + Pillow (image script only), Gradle (Android resources).

**Spec:** `docs/specs/2026-11-24-coup-noir-neon-upgrade-design.md` (Part 1 and the "Launch screen" section).

**This is plan 1 of 4.** Plans 2 (sounds), 3 (quick effects) and 4 (scenes) are written after this one has been tested on the phone.

### Task 2: Decorative images (haze and card shading)

**Files:**
- Create: `scripts/generate-images.py`
- Create (generated): `assets/images/haze.png`, `assets/images/card-shade.png`

- [ ] **Step 1: Write the script**

Create `scripts/generate-images.py`:

```python
"""
Makes the two decorative images of the Noir Neon look.

    python -I scripts/generate-images.py

  assets/images/haze.png        512x512: a soft cyan glow fading to transparent at the edge.
                                Laid over the black background in the top-left corner.
  assets/images/card-shade.png  16x128: light at the top, clear in the middle, dark at the
                                bottom. Stretched over a coloured card it turns the flat fill
                                into a gradient of the same colour.
Needs Pillow and NumPy.
"""

from pathlib import Path

import numpy as np
from PIL import Image

ROOT = Path(__file__).resolve().parent.parent
OUT = ROOT / "assets" / "images"
CYAN = (0, 229, 255)


def haze() -> Image.Image:
    size = 512
    y, x = np.mgrid[0:size, 0:size]
    centre = (size - 1) / 2
    distance = np.sqrt((x - centre) ** 2 + (y - centre) ** 2) / centre  # 0 centre .. 1 edge
    alpha = np.clip(1 - distance, 0, 1) ** 2 * 0.16 * 255
    rgba = np.zeros((size, size, 4), dtype=np.uint8)
    rgba[..., 0], rgba[..., 1], rgba[..., 2] = CYAN
    rgba[..., 3] = alpha.astype(np.uint8)
    return Image.fromarray(rgba, "RGBA")


def card_shade() -> Image.Image:
    width, height = 16, 128
    t = np.linspace(0, 1, height)
    rgba = np.zeros((height, width, 4), dtype=np.uint8)
    for row, v in enumerate(t):
        if v < 0.5:  # white highlight fading out towards the middle
            colour, alpha = 255, (0.5 - v) / 0.5 * 0.22
        else:  # black shadow growing towards the bottom
            colour, alpha = 0, (v - 0.5) / 0.5 * 0.38
        rgba[row, :, 0:3] = colour
        rgba[row, :, 3] = int(alpha * 255)
    return Image.fromarray(rgba, "RGBA")


def main() -> None:
    OUT.mkdir(parents=True, exist_ok=True)
    haze().save(OUT / "haze.png", optimize=True)
    card_shade().save(OUT / "card-shade.png", optimize=True)
    print(f"Wrote {OUT / 'haze.png'} and {OUT / 'card-shade.png'}")


if __name__ == "__main__":
    main()
```

import React from 'react';
import { Image, StyleSheet, useWindowDimensions, View } from 'react-native';

const HAZE = require('../../assets/images/haze.png');

/**
 * The Noir Neon background: black with a faint cyan haze in the top-left corner,
 * like the light behind the man in the intro. Decoration only: it takes no touches
 * and is hidden from screen readers.
 */
export function Backdrop() {
  const { width, height } = useWindowDimensions();
  const size = Math.max(width, height) * 0.9;
  return (
    <View
      testID="backdrop"
      pointerEvents="none"
      accessible={false}
      importantForAccessibility="no-hide-descendants"
      style={styles.fill}
    >
      <Image
        source={HAZE}
        resizeMode="stretch"
        fadeDuration={0}
        style={{
          position: 'absolute',
          width: size,
          height: size,
          left: -size * 0.35,
          top: -size * 0.45,
        }}
      />
    </View>
  );
}

const styles = StyleSheet.create({
  fill: { ...StyleSheet.absoluteFillObject, overflow: 'hidden' },
});
```

/**
 * Two scripted players join the lobby whose code is in BOT_CODE and play simply:
 * Income on their turn, pass on every prompt, give up their first card when asked.
 * They stop after BOT_MINUTES (default 10) or when the game is over, then leave.
 *
 *   BOT_CODE=ABCDE npx jest --config jest.bots.config.js
 */
import { deleteApp, initializeApp } from 'firebase/app';
import { getDatabase } from 'firebase/database';
import { pendingResponders } from '../../src/engine/rules';
import { Game } from '../../src/engine/types';
import { firebaseConfig } from '../../src/firebase/config';
import { createGameService } from '../../src/firebase/gameService';

const code = (process.env.BOT_CODE ?? '').trim().toUpperCase();
const minutes = Number(process.env.BOT_MINUTES ?? '10');

function bot(name: string) {
  const app = initializeApp(firebaseConfig, `bot-${name}`);
  const service = createGameService(getDatabase(app));
  const playerId = `bot-${name.toLowerCase()}-${Date.now()}`;
  return { app, service, playerId, name };
}

async function act(b: ReturnType<typeof bot>, gameId: string, game: Game) {
  const { state } = game;
  const me = b.playerId;
  if (state.phase === 'action' && state.currentTurnPlayer === me) {
    await b.service.dispatch(gameId, { type: 'income', playerId: me });
  } else if (
    (state.phase === 'awaitingResponses' || state.phase === 'awaitingBlockResponses') &&
    pendingResponders(game).includes(me)
  ) {
    await b.service.dispatch(gameId, { type: 'pass', playerId: me, seq: state.claimSeq });
  } else if (state.phase === 'loseInfluence' && state.pending?.loseInfluence?.playerId === me) {
    const index = game.players[me].influence.findIndex(card => !card.revealed);
    await b.service.dispatch(gameId, { type: 'loseInfluence', playerId: me, cardIndex: index });
  }
}

it('plays as two extra players', async () => {
  expect(code).toMatch(/^[A-Z0-9]{5}$/);
  const bots = [bot('Ana'), bot('Nikos')];
  const gameIds = await Promise.all(
    bots.map(b => b.service.joinGame(code, b.playerId, b.name)),
  );
  const stops: Array<() => void> = [];
  await new Promise<void>(resolve => {
    const end = setTimeout(resolve, minutes * 60 * 1000);
    bots.forEach((b, i) => {
      let busy = false;
      stops.push(
        b.service.subscribe(
          gameIds[i],
          game => {
            if (!game || game.status === 'finished') {
              clearTimeout(end);
              resolve();
              return;
            }
            if (busy) return;
            busy = true;
            act(b, gameIds[i], game)
              .catch(error => console.warn(`${b.name}: ${error}`))
              .finally(() => {
                busy = false;
              });
          },
          error => console.warn(`${b.name}: ${error}`),
        ),
      );
    });
  });
  stops.forEach(stop => stop());
  await Promise.all(bots.map(b => deleteApp(b.app)));
});
```
